#include <iostream>
using namespace std;

int stk[10], maxsize=10, top=0;
bool isEmpty();
bool isFull();
bool pop();
bool push(int i);
void TopElement();
void show();

int main ()
{
    push(1);
    push(4);
    push(5);
    push(6);
    show();
    cout << endl;
    pop();
    pop();
    pop();
    show();
    return 0;
}
bool isEmpty()
{
    return (top==0);
}
bool isFull()
{
    return (top==maxsize);
}
bool push(int i)
{
    if(isFull())
    {
        return false;
    }
    stk[top++]=i;
    return true;
}
bool pop()
{
    if(isEmpty())
    {
        return false;
    }
    top--;
    return true;

}
void show()
{
    for(int i=top-1; i>=0; i--)
    {
        cout << stk[i] << " ";
    }
}
void TopElement()
{
    cout << "Topmost element:  "<<stk[top-1] << endl;
}
