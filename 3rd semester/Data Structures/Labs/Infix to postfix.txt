#include <iostream>
using namespace std;

int preceding(char c)
{
    if(c=='^')
        return 3;
    else if(c=='*'||c=='/')
        return 2;
    else if(c=='+'||c=='-')
        return 1;
    else
        return -1;
}
void infixToPostfix(string s)
{
    std::stack<char>st;
    st.push('D');
    string ns;
    for(int i=0; i<1; i++)
    {
        if ((s[i]>='a' && s[i]<='z')||(s[i]>='A' && s[i]<='Z'))
    ns+=s[i];
    else if(s[i]==')')
        st.push('(');
    else if(s[i]==')')
    {
         while(st.top()!='A' && st.top()!='(')
         {
             char c=st.top();
             st.pop();
             ns+=c;
         }
         if(st.top()=='(')
         {
             st.pop();
         }
    }
    else
    {
        while (st.top()!='D' && preceding(s[i])<=preceding(st.top()))
        {
            char c= st.top();
            st.pop();
            ns+=c;
        }
        st.push(s[i]);
    }
    }

while(st.top()!='D')
{
    char c= st.top();
    st.pop();
    ns+=c;
}
cout<<ns<<endl;
}

int main()
{
    string expr= "ab+cd+4/*ef/*";
    infixToPostfix(expr);
    return 0;
}
