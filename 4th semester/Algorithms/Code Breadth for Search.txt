#include <iostream>
#include <cstdlib>
#include <vector>
#include <list>
#include <queue>

using namespace std;

enum Color {White = 1, Gray, Black};                   //Restricting variable 'Color'

struct node//creating a structure named node
{
    int vertex;
    Color curr_color;
    int discovery_time;
    list<int> *adj = new list<int>;
};

vector<node> Node_list;
void makeAdjacency(int num_of_vertex)
{
    cout<< "Enter Adjacency Lists\n";
    cout<< "****** Enter a '-1' to end the list of a node *******"<< endl;
    for(int i=0;i<num_of_vertex;i++)             //run the loop once for each
    {
        cout<< "Vertex " << i << endl;
        int adjacent;
        while(1)            //constant time loop
        {
            cin>> adjacent;
            if(adjacent == -1)
                break;
            Node_list[i].adj->push_back(adjacent);              //push the adjacent node to the adjacency list
        }
        cout<< endl;
    }
}

void printAdjacency(int num_of_vertex)
{
    cout<< " Adjacency List "<<endl;
    cout<< "Vertex adjacent"<<endl;
    for(int i=0;i<num_of_vertex;i++)
    {
        cout<< i << "\t [";
        {
        for(list<int>::iterator it = Node_list[i].adj->begin(); it != Node_list[i].adj->end(); it++)
            cout<< *it << ",";                 //print the value of the iterator
        }
        cout<< "]" << endl;
    }
}

void BFS(int startNode)
{
    queue<node> Q;             // declaring node type queue

    Q.push(Node_list[startNode]);               //push the node into the queue
    node currentNode;              //declaring a object of type node

    while(!Q.empty())                //stop running the loop when the queue is empty
    {
        currentNode = Q.front();             //insert the node into the queue
        for(list<int>::iterator it=currentNode.adj->begin(); it != currentNode.adj->end(); it++)             //run the loop according to the adjacencies of a particular vertex
        {
            if(Node_list[*it].curr_color == White)            //run when the node is not visited
            {
                Node_list[*it].curr_color = Gray;            //turn the color of the node into gray
                Node_list[*it].discovery_time = currentNode.discovery_time + 1;            //initialize the discovery time
                Q.push(Node_list[*it]);              //push the node to the back of the queue
            }
        }

        cout<< "Current Node: "<< currentNode.vertex << endl;
        cout<< "Discovery Time: "<< currentNode.discovery_time << endl;
        currentNode.curr_color = Black;//turn the color of the explored node into black
        Q.pop();//remove the node from the queue
    }
}

int main()
{
    int num_of_vertex;
    int i;
    cout<< "Enter number of vertex: ";
    cin>> num_of_vertex;

    for(i=0;i<num_of_vertex;i++)
    {
        node tmp;
        tmp.vertex = i;
        tmp.curr_color = White;
        tmp.discovery_time = 0;
        Node_list.push_back(tmp);
    }
    makeAdjacency(num_of_vertex);            //calling function
    printAdjacency(num_of_vertex);          //calling function

    cout<< "Enter the starting node for BFS visit"<<endl;
    int startNode;
    cin>> startNode;
    BFS(startNode);            //function calling
    return 0;
}
