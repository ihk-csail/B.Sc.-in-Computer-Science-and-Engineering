class Edge       //declaring a class called 'Edge'
{

};

void searchVertex(Edge *ar,char *arr)
{
    int i,j;
    cout<<"The adjacent vertices are :"<<endl;
    for(i=0;i<8;i++)          //search the adjacency list to find the adjacent node
     {
        cout<<arr[i]<<" -> ";
        for(j=0;j<10;j++)            // searching the adjacency list
        {
            if(ar[j].startnode==arr[i])           //print the endnode under given condition
            {
                cout<<ar[j].endnode<<" ";
            }
            if(ar[j].endnode==arr[i])              //print the startnode under given condition
            {
                cout<<ar[j].startnode<<" ";
            }

        }
        cout<<endl;
    }
}

        }
        else if(ar[i].endnode==c)         //print the startnode under given condition

}
